STANDARDS

PREFIX dct: <http://purl.org/dc/terms/>
PREFIX dcat: <http://www.w3.org/ns/dcat#>
PREFIX adms: <http://www.w3.org/ns/adms#>

SELECT ?standard ?title
WHERE {
  ?standard a dct:Standard ;
            dct:title ?title .
  FILTER (lang(?title) = "en")
}
ORDER BY ?title
